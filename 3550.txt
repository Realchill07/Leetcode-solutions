class Solution {
    public int smallestIndex(int[] nums) {
        int len = nums.length - 1;
        int sum = 0;
        for(int i = 0; i <= len; i++){
            sum = 0;
            if(nums[0] == 0 && i == 0) return 0;
            while(nums[i] > 0){
                int a = nums[i]%10;
                sum += a;
                nums[i] /= 10;
            }
            if(sum == i) return sum;

        }

        return -1;
    }
}
